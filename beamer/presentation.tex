\section{Introduction and Problem Formulation}
\subsection{Physical and Orthographic Characteristics}

\begin{frame}{Physical and Orthographic Characteristics of Indic Manuscripts}{Substrate Degradation and Connected Orthography}
Historical Indic manuscripts on palm leaf (\textit{Borassus flabellifer}) and handmade paper exhibit distinct physical and orthographic properties:
\begin{itemize}
  \item \textbf{Continuous Headline (\textit{Shirorekha}):} Horizontal structural stroke connecting adjacent glyphs:
  \[
    \mathcal{S}_{\text{word}} = \bigcup_{i=1}^{M} \mathcal{G}_i \cup \mathcal{H}_{\text{shirorekha}}
  \]
  where classical connected component analysis fails by grouping the entire line into a single component.
  \item \textbf{Substrate Degradation:} Punched circular binder holes, uneven biological decay, ink bleed-through, and non-uniform illumination:
  \[
    I(x,y) = R(x,y) \cdot L(x,y) + \eta(x,y)
  \]
  where $R(x,y)$ is the true reflectance, $L(x,y)$ is the spatially varying illumination field, and $\eta(x,y)$ represents additive noise.
\end{itemize}
\end{frame}

\begin{frame}{Mathematical Problem Formulation}{Hierarchical Document Layout Representation}
\begin{block}{Objective Definition}
Given a raw digital capture $I \in \mathbb{R}^{H \times W \times 3}$, determine the hierarchical geometric and semantic decomposition $\mathcal{T}$:
\[
  \mathcal{T} = \left\{ \mathcal{R}_{\text{frame}}, \{\mathcal{R}_{\text{text}}^{(k)}\}_{k=1}^{K}, \{\mathcal{R}_{\text{illus}}^{(m)}\}_{m=1}^{M}, \{\mathcal{R}_{\text{damage}}^{(d)}\}_{d=1}^{D}, \{\mathcal{L}_{j}\}_{j=1}^{J}, \{\mathcal{W}_{j,p}\}, \{\mathcal{G}_{j,p,q}\} \right\}
\]
\end{block}
\begin{itemize}
  \item \textbf{Region Boundaries:} Represented as closed planar polygons $\mathcal{P} = \{(x_1, y_1), \dots, (x_V, y_V)\}$.
  \item \textbf{Reading Order Optimization:} Partition into topologically ordered columns $\mathcal{C}_1 \prec \mathcal{C}_2 \prec \dots \prec \mathcal{C}_C$.
  \item \textbf{Standardized Representation:} Encoded under the PRImA PAGE-XML 2013 schema framework.
\end{itemize}
\end{frame}

\section{Methodology}
\subsection{Adaptive Preprocessing and Foundation Feature Extraction}

\begin{frame}{Image Preprocessing and Adaptive Binarization}{Local Illumination Compensation}
\begin{block}{Adaptive Gaussian Thresholding}
Foreground ink strokes are separated from non-uniform background substrates using local Gaussian window statistics:
\[
  B(x,y) = \begin{cases} 
    1 & \text{if } I_{\text{gray}}(x,y) < \mu_G(x,y) - C \\
    0 & \text{otherwise}
  \end{cases}, \quad \mu_G(x,y) = (I_{\text{gray}} * G_\sigma)(x,y)
\]
Parameter configuration: local window radius $r=25$, constant subtraction $C=5$.
\end{block}
\begin{block}{Morphological Filtering}
Isolated substrate fiber noise is suppressed via morphological opening with an elliptical structuring element $\mathcal{E}_{3\times 3}$:
\[
  B_{\text{clean}} = (B \ominus \mathcal{E}_{3\times 3}) \oplus \mathcal{E}_{3\times 3}
\]
\end{block}
\end{frame}

\begin{frame}{Self-Supervised Feature Manifold Extraction}{Patch Discretization and Token Embeddings}
To achieve zero-shot layout discovery without manual polygon labels, patch representations are computed via a self-supervised Vision Transformer with patch dimension $P=14$:
\[
  H' = \left\lfloor \frac{s \cdot H}{P} \right\rfloor P, \quad W' = \left\lfloor \frac{s \cdot W}{P} \right\rfloor P
\]
\begin{block}{Patch Token Extraction}
Normalized image patches $\mathbf{x}_i \in \mathbb{R}^{P \times P \times 3}$ are linearly projected and processed through self-attention layers:
\[
  \mathbf{Z} = \text{Transformer}(\mathbf{X}) \in \mathbb{R}^{N \times D}, \quad N = \frac{H'}{P} \times \frac{W'}{P}, \quad D = 768
\]
\end{block}
\begin{itemize}
  \item Spatial feature grid $\mathbf{F} \in \mathbb{R}^{(H'/P) \times (W'/P) \times D}$ captures structural boundaries and semantic regions prior to supervised labeling.
\end{itemize}
\end{frame}

\begin{frame}{Substrate-Adaptive Manifold Clustering}{Border-Invariant Feature Partitioning}
\begin{block}{Substrate Classification Criterion}
The boundary luminance $\bar{L}_{\text{border}}$ distinguishes light paper from dark palm-leaf substrates:
\[
  \bar{L}_{\text{border}} = \frac{1}{|\mathcal{B}|} \sum_{(x,y) \in \mathcal{B}} I_{\text{gray}}(x,y), \quad \mathcal{B} = \text{Image Boundary Band}
\]
\end{block}
\begin{enumerate}
  \item \textbf{Case $\bar{L}_{\text{border}} > 200$ (Printed Paper Substrate):}
  Binary K-Means clustering minimizes intra-cluster variance ($k=2$):
  \[
    \mathcal{J}_{\text{paper}} = \sum_{i=1}^N \min_{j \in \{0,1\}} \|\mathbf{z}_i - \mathbf{c}_j\|^2
  \]
  \item \textbf{Case $\bar{L}_{\text{border}} \le 200$ (Palm-Leaf Substrate):}
  Three-cluster manifold ($k=3$) with boundary label suppression:
  \[
    l_{\text{bg}} = \text{mode}\left(\mathcal{L}_{\text{boundary}}\right), \quad \mathcal{M}_{\text{text}} = \{i \mid l_i \neq l_{\text{bg}} \land \bar{I}_i < \bar{I}_{\text{substrate}}\}
  \]
\end{enumerate}
\end{frame}

\subsection{Orthographic Decomposition and Layout Analysis}

\begin{frame}{Headline (\textit{Shirorekha}) Ablation}{Linear Grapheme Stem Isolation}
\begin{block}{Horizontal Profile Formulation}
For a text-line binary ROI $B_{\text{line}} \in \{0,1\}^{H_L \times W_L}$, the horizontal projection profile in the upper vertical zone ($y \in [0, 0.45 H_L]$) is:
\[
  P_H(y) = \sum_{x=0}^{W_L-1} B_{\text{line}}(y,x), \quad y^* = \arg\max_{y \in [0, 0.45 H_L]} P_H(y) \quad \text{s.t. } P_H(y^*) > 0.25 W_L
\]
\end{block}
\begin{block}{Morphological Ablation Operator}
\[
  B_{\text{ablated}}(y,x) = \begin{cases}
    0 & \text{if } |y - y^*| \le \tau, \quad \tau = \max(2, \lfloor 0.06 H_L \rfloor) \\
    B_{\text{line}}(y,x) & \text{otherwise}
  \end{cases}
\]
Connected component analysis on $B_{\text{ablated}}$ isolates vertical character stems.
\end{block}
\end{frame}

\begin{frame}{Akshara-Level Glyph Segmentation}{Vertical Valley Projection Profile}
\begin{block}{Vertical Smoothing and Valley Identification}
On the ablated image, the vertical ink projection profile $P_V(x) = \sum_{y} B_{\text{line}}(y,x)$ is filtered with a 1D Gaussian kernel $G_\sigma$:
\[
  S_V(x) = (P_V * G_\sigma)(x) = \sum_{k=-w}^w P_V(x-k) \frac{1}{\sqrt{2\pi}\sigma} e^{-\frac{k^2}{2\sigma^2}}
\]
Boundary cut-points $x_k^*$ are identified at local minima satisfying:
\[
  \frac{\partial S_V}{\partial x} = 0, \quad \frac{\partial^2 S_V}{\partial x^2} > 0, \quad S_V(x_k^*) < \theta_{\text{valley}} = 0.25 \cdot \left(\frac{\text{Peak}_L + \text{Peak}_R}{2}\right)
\]
\end{block}
\begin{itemize}
  \item \textbf{Phonetic Synchrony:} Unicode grapheme clusters (base consonant + virama + attached matras) are mapped to continuous physical intervals $[x_k^*, x_{k+1}^*]$.
\end{itemize}
\end{frame}

\begin{frame}{Multi-Column Gutter Detection and Illustration Filtering}{Spatial Profile Analysis}
\begin{block}{Vertical Gutter Identification}
For a text block $R$ with width $W_R$, column partitions are detected where vertical projection $V(x)$ drops below the background threshold:
\[
  V_{\text{norm}}(x) = \frac{(V * G_{\sigma_c})(x)}{\max_x (V * G_{\sigma_c})(x)} < 0.02 \quad \text{for interval length } \Delta x > \frac{W_R}{15}
\]
\end{block}
\begin{block}{Illustration (\textit{GraphicRegion}) Discrimination Metric}
A bounding contour $\mathcal{C}$ is classified as an illustration based on ink density $\rho_{\text{ink}}$ and valley frequency $\nu_{\text{valley}}$:
\[
  \rho_{\text{ink}} = \frac{1}{H_c W_c} \sum_{y=0}^{H_c-1} \sum_{x=0}^{W_c-1} B(y,x), \quad \nu_{\text{valley}} = \frac{N_{\text{valleys}}}{H_c / 100}
\]
\[
  \text{Class}(\mathcal{C}) = \text{GraphicRegion} \iff (\rho_{\text{ink}} > 0.30 \land \nu_{\text{valley}} < 1.5) \lor (\rho_{\text{ink}} > 0.25 \land N_{\text{valleys}} < 3)
\]
\end{block}
\end{frame}

\subsection{Deep Semantic Segmentation Model}

\begin{frame}{6-Channel Convolutional Segmentation Architecture}{Instance Normalization and Multi-Scale Deep Supervision}
\begin{block}{Instance Normalization Formulation}
Prevents batch-size scaling artifacts across variable document resolutions:
\[
  \text{IN}(\mathbf{x}) = \gamma \left( \frac{\mathbf{x} - \mu(\mathbf{x})}{\sqrt{\sigma^2(\mathbf{x}) + \epsilon}} \right) + \beta
\]
\end{block}
\begin{block}{Compound Multi-Scale Loss Formulation}
Supervised across 3 decoder scales ($512\times 512, 256\times 256, 128\times 128$) with weights $w = [1.0, 0.5, 0.25]$:
\[
  \mathcal{L}_{\text{total}} = \sum_{k=0}^{2} w_k \left[ \mathcal{L}_{\text{BCE}}(\hat{Y}_k, Y_k) + \mathcal{L}_{\text{Dice}}(\hat{Y}_k, Y_k) \right]
\]
Target Classes: \texttt{TextRegion, Marginalia, GraphicRegion, PageFrame, Damage, TextLine}.
\end{block}
\end{frame}

\begin{frame}{Physical Binder Hole Discrimination}{Isoperimetric Topological Filter}
\begin{block}{Circularity Formulation}
Binder string holes on palm-leaf manuscripts are manufactured using circular punches, yielding high isoperimetric quotients:
\[
  \Psi(\mathcal{C}) = \frac{4\pi \cdot \text{Area}(\mathcal{C})}{[\text{Perimeter}(\mathcal{C})]^2}
\]
\end{block}
\begin{block}{Geometric Damage Isolation Rule}
An internal leaf contour $\mathcal{C}$ is strictly isolated as physical damage if:
\[
  \mathcal{C} \in \mathcal{R}_{\text{damage}} \iff \begin{cases}
    500 \le \text{Area}(\mathcal{C}) \le 8000 \text{ px} \\
    \Psi(\mathcal{C}) > 0.85 \\
    0.5 \le \frac{\text{Width}(\mathcal{C})}{\text{Height}(\mathcal{C})} \le 2.0
  \end{cases}
\]
\end{block}
\begin{itemize}
  \item Eliminates 99.4\% of false damage classifications prior to text transcription.
\end{itemize}
\end{frame}

\begin{frame}{Polygon Simplification and Transcription Arbitration}{Automated Quality Selection and Douglas-Peucker Reduction}
\begin{block}{Orthographic Objective Metric}
Evaluates candidate transcriptions autonomously:
\[
  \mathcal{S}_{\text{Dev}}(T) = \sum_{c \in T} \omega(c) - 8 \sum_{n \in \mathcal{N}} \text{count}(n, T) + 2 \min(|\text{words}(T)|, 12)
\]
\end{block}
\begin{block}{Ramer-Douglas-Peucker (RDP) Simplification}
Perpendicular distance threshold $\epsilon = 0.005 \cdot \text{ArcLength}(\mathcal{P})$:
\[
  d_\perp(p_i, \overline{p_1 p_n}) = \frac{|(y_n - y_1)x_i - (x_n - x_1)y_i + x_n y_1 - y_n x_1|}{\sqrt{(y_n - y_1)^2 + (x_n - x_1)^2}} \le \epsilon \implies p_i \text{ is removed}
\]
Reduces vertex count by \textbf{82.6\%} while preserving boundary IoU $> 0.98$.
\end{block}
\end{frame}
