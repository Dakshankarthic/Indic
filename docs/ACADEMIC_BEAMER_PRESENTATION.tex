\section{Introduction and Problem Formulation}

\begin{frame}{Introduction \& Problem Statement}{Heritage Preservation Bottlenecks in Indic Manuscripts}
\begin{columns}
  \begin{column}{0.48\textwidth}
    \begin{block}{Heritage at Risk \& OCR Limitations}
      \begin{itemize}
        \item Thousands of Indic manuscripts (\textit{Ramcharitmanas}, palm-leaf texts) suffer biological decay, mold, and ink corrosion.
        \item Classical OCR tools fail on complex Devanagari with conjuncts, matras, and continuous headlines:
        \[
          I(x,y) = R(x,y) \cdot L(x,y) + \eta(x,y)
        \]
      \end{itemize}
    \end{block}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{block}{Manual Annotation Bottleneck}
      \begin{itemize}
        \item Expert paleographers spend 30 to 60 minutes per page marking regions, lines, words, and characters.
        \item For collections of 3,000+ pages, manual annotation requires over 2,000 human-hours.
        \item Lack of a unified pipeline integrating layout, recognition, and granular segmentation.
      \end{itemize}
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Mathematical Problem Formulation}{Hierarchical Document Layout and Semantic Partitioning}
\begin{block}{Objective Definition (PRImA PAGE-XML 2013 Hierarchy)}
Given raw capture $I \in \mathbb{R}^{H \times W \times 3}$, determine hierarchical geometric and semantic decomposition $\mathcal{T}$:
\[
  \mathcal{T} = \left\{ \mathcal{R}_{\text{frame}}, \{\mathcal{R}_{\text{text}}^{(k)}\}_{k=1}^{K}, \{\mathcal{R}_{\text{illus}}^{(m)}\}_{m=1}^{M}, \{\mathcal{R}_{\text{damage}}^{(d)}\}_{d=1}^{D}, \{\mathcal{L}_{j}\}_{j=1}^{J}, \{\mathcal{W}_{j,p}\}, \{\mathcal{G}_{j,p,q}\} \right\}
\]
\end{block}
\begin{itemize}
  \item \textbf{Region Boundaries:} Closed planar polygonal chains $\mathcal{P} = \{(x_1, y_1), \dots, (x_V, y_V)\}$.
  \item \textbf{Reading Order Optimization:} Partition into topologically ordered columns $\mathcal{C}_1 \prec \mathcal{C}_2 \prec \dots \prec \mathcal{C}_C$.
  \item \textbf{Continuous Headline Union:} $\mathcal{S}_{\text{word}} = \bigcup_{i=1}^M \mathcal{G}_i \cup \mathcal{H}_{\text{shirorekha}}$.
\end{itemize}
\end{frame}

\subsection{Stage 1: Layout Detection and Geometry}

\begin{frame}{Image Preprocessing and Adaptive Binarization}{Local Illumination Compensation}
\begin{block}{Adaptive Gaussian Thresholding}
Foreground ink strokes are separated from non-uniform background substrates using local Gaussian window statistics:
\[
  B(x,y) = \begin{cases} 
    1 & \text{if } I_{\text{gray}}(x,y) < \mu_G(x,y) - C \\
    0 & \text{otherwise}
  \end{cases}, \quad \mu_G(x,y) = (I_{\text{gray}} * G_\sigma)(x,y)
\]
Parameter configuration: local window radius $r=25$, constant subtraction $C=5$.
\end{block}
\begin{block}{Morphological Filtering}
Isolated substrate fiber noise is suppressed via morphological opening with an elliptical structuring element $\mathcal{E}_{3\times 3}$:
\[
  B_{\text{clean}} = (B \ominus \mathcal{E}_{3\times 3}) \oplus \mathcal{E}_{3\times 3}
\]
\end{block}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 7: Stage 1: DINOv2 Feature Geometry & Self-Attention
% ─────────────────────────────────────────────────────────────
\begin{frame}{Stage 1: Foundation Vision Transformer (DINOv2)}{Patch Embedding, Multi-Head Self-Attention, and Zero-Shot Manifolds}
\begin{block}{Patch Discretization & Position Encoding}
Model Architecture: \texttt{DINOv2-ViT-B/14} ($D = 768$, $P = 14$). Image dimensions aligned to patch multiples:
\[
  \mathbf{x}_0 = \left[ \mathbf{x}_{\text{cls}}; \, \mathbf{x}_p^1 \mathbf{E}; \, \dots; \, \mathbf{x}_p^N \mathbf{E} \right] + \mathbf{E}_{\text{pos}}, \quad N = \frac{H'}{14} \times \frac{W'}{14}, \quad \mathbf{E} \in \mathbb{R}^{588 \times 768}
\]
\end{block}
\begin{block}{Multi-Head Self-Attention (MHSA) Formulation}
Scaled dot-product pairwise affinity between patches across 12 attention heads ($d_k = 64$):
\[
  \text{Attention}(\mathbf{Q}, \mathbf{K}, \mathbf{V}) = \text{softmax}\left( \frac{\mathbf{Q} \mathbf{K}^T}{\sqrt{d_k}} \right) \mathbf{V}, \quad A_{ij} = \frac{\mathbf{z}_i \cdot \mathbf{z}_j}{\|\mathbf{z}_i\| \|\mathbf{z}_j\|} = \cos(\theta_{ij})
\]
\end{block}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 8: Stage 1: Substrate-Adaptive Clustering
% ─────────────────────────────────────────────────────────────
\begin{frame}{Stage 1: Substrate-Adaptive Manifold Clustering}{Border-Invariant Feature Partitioning}
\begin{block}{Substrate Classification Criterion}
The boundary luminance $\bar{L}_{\text{border}}$ distinguishes light paper from dark palm-leaf substrates:
\[
  \bar{L}_{\text{border}} = \frac{1}{|\mathcal{B}|} \sum_{(x,y) \in \mathcal{B}} I_{\text{gray}}(x,y), \quad \mathcal{B} = \text{Image Boundary Band}
\]
\end{block}
\begin{enumerate}
  \item \textbf{Case $\bar{L}_{\text{border}} > 200$ (Printed Paper Substrate, $k=2$):}
  Binary K-Means clustering minimizes intra-cluster variance:
  \[
    \mathcal{J}_{\text{paper}} = \sum_{i=1}^N \min_{j \in \{0,1\}} \|\mathbf{z}_i - \mathbf{c}_j\|^2
  \]
  \item \textbf{Case $\bar{L}_{\text{border}} \le 200$ (Palm-Leaf Substrate, $k=3$):}
  Three-cluster manifold with boundary label mode suppression:
  \[
    l_{\text{bg}} = \text{mode}\left(\mathcal{L}_{\text{boundary}}\right), \quad \mathcal{M}_{\text{text}} = \{i \mid l_i \neq l_{\text{bg}} \land \bar{I}_i < \bar{I}_{\text{substrate}}\}
  \]
\end{enumerate}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 9: Stage 1: Column Parsing & Illustration Filtering
% ─────────────────────────────────────────────────────────────
\begin{frame}{Stage 1: Column Parsing \& Illustration Discrimination}{Spatial Profile Analysis}
\begin{block}{Vertical Gutter Identification}
For a text block $R$ with width $W_R$, column partitions are detected where vertical projection $V(x)$ drops below the background threshold:
\[
  V_{\text{norm}}(x) = \frac{(V * G_{\sigma_c})(x)}{\max_x (V * G_{\sigma_c})(x)} < 0.02 \quad \text{for interval length } \Delta x > \frac{W_R}{15}
\]
\end{block}
\begin{block}{Illustration (\textit{GraphicRegion}) Discrimination Metric}
A bounding contour $\mathcal{C}$ is classified as an illustration based on ink density $\rho_{\text{ink}}$ and valley frequency $\nu_{\text{valley}}$:
\[
  \rho_{\text{ink}} = \frac{1}{H_c W_c} \sum_{y=0}^{H_c-1} \sum_{x=0}^{W_c-1} B(y,x), \quad \nu_{\text{valley}} = \frac{N_{\text{valleys}}}{H_c / 100}
\]
\[
  \text{Class}(\mathcal{C}) = \text{GraphicRegion} \iff (\rho_{\text{ink}} > 0.30 \land \nu_{\text{valley}} < 1.5) \lor (\rho_{\text{ink}} > 0.25 \land N_{\text{valleys}} < 3)
\]
\end{block}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 10: Stage 1: 6-Channel nnU-Net Deep Segmentation
% ─────────────────────────────────────────────────────────────
\begin{frame}{Stage 1: 6-Channel Deep Semantic Segmentation}{Instance Normalization and Multi-Scale Deep Supervision}
\begin{block}{Instance Normalization & Feature Propagation}
\[
  \mathbf{x}^{(l)} = \text{LeakyReLU}\left( \text{IN}\left( \mathbf{W}_2 * \text{LeakyReLU}\left( \text{IN}\left( \mathbf{W}_1 * \mathbf{x}^{(l-1)} \right) \right) \right) \right), \quad \text{IN}(\mathbf{x}) = \gamma \left( \frac{\mathbf{x} - \mu(\mathbf{x})}{\sqrt{\sigma^2(\mathbf{x}) + \epsilon}} \right) + \beta
\]
\end{block}
\begin{block}{Multi-Scale Compound Focal-Dice Loss}
Supervised across 3 decoder scales ($512\times 512, 256\times 256, 128\times 128$) with weights $w = [1.0, 0.5, 0.25]$:
\[
  \mathcal{L}_{\text{total}} = \sum_{k=0}^{2} w_k \left[ \mathcal{L}_{\text{Focal}}(\hat{Y}_k, Y_k) + \mathcal{L}_{\text{Dice}}(\hat{Y}_k, Y_k) \right]
\]
Channels: \texttt{TextRegion, Marginalia, GraphicRegion, PageFrame, Damage, TextLine}.
\end{block}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 11: Stage 2: Text Recognition (OCR / HTR Engine)
% ─────────────────────────────────────────────────────────────
\subsection{Stage 2: Text Recognition Engine}

\begin{frame}{Stage 2: Multilingual Text Recognition Engine}{Recurrent Sequence Modeling and Connectionist Temporal Classification}
\begin{block}{CRNN Feature Sequence Extraction}
Line crop slices $\mathbf{x}_t$ encoded via Bidirectional LSTM context (Surya OCR v2 reference with JSON text cache for 1,054 benchmark folios):
\[
  \mathbf{h}_t = \left[ \overrightarrow{\text{LSTM}}(\mathbf{x}_t); \, \overleftarrow{\text{LSTM}}(\mathbf{x}_t) \right] \in \mathbb{R}^{2 D_{\text{hidden}}}, \quad \mathbf{y}_t = \text{softmax}(\mathbf{W}_o \mathbf{h}_t + \mathbf{b}_o)
\]
\end{block}
\begin{block}{Connectionist Temporal Classification (CTC) Decoding}
\[
  P(\mathbf{Y} \mid \mathbf{X}) = \sum_{\boldsymbol{\pi} \in \mathcal{B}^{-1}(\mathbf{Y})} \prod_{t=1}^T y_{\pi_t}^t, \quad \hat{\mathbf{Y}} = \arg\max_{\mathbf{Y}} \left[ \log P_{\text{CTC}}(\mathbf{Y} \mid \mathbf{X}) + \alpha \log P_{\text{LM}}(\mathbf{Y}) + \beta |\mathbf{Y}| \right]
\]
\end{block}
\begin{itemize}
  \item $P_{\text{LM}}(\mathbf{Y})$ enforces Sanskrit grammatical constraints and valid Brahmic root morphology.
\end{itemize}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 12: Stage 3: Shirorekha Ablation & Akshara Splitting
% ─────────────────────────────────────────────────────────────
\subsection{Stage 3: Granular Extraction}

\begin{frame}{Stage 3: Granular Word \& Character (Akshara) Splitting}{Headline (\textit{Shirorekha}) Ablation and Gaussian Valley Tracking}
\begin{block}{Horizontal Profile Formulation & Ablation Operator}
\[
  P_H(y) = \sum_{x=0}^{W_L-1} B_{\text{line}}(y,x), \quad y^* = \arg\max_{y \in [0, 0.45 H_L]} P_H(y) \quad \text{s.t. } P_H(y^*) > 0.25 W_L
\]
\[
  B_{\text{ablated}}(y,x) = \begin{cases} 0 & \text{if } |y - y^*| \le \tau, \quad \tau = \max(2, \lfloor 0.06 H_L \rfloor) \\ B_{\text{line}}(y,x) & \text{otherwise} \end{cases}
\]
\end{block}
\begin{block}{1D Gaussian Valley Cut-Point Detection}
\[
  S_V(x) = (P_V * G_\sigma)(x), \quad \frac{\partial S_V}{\partial x} = 0, \quad \frac{\partial^2 S_V}{\partial x^2} > 0, \quad S_V(x_k^*) < \theta_{\text{valley}}
\]
\end{block}
\end{frame}

% ─────────────────────────────────────────────────────────────
% FRAME 13: Physical Binder Hole Discrimination
% ─────────────────────────────────────────────────────────────
\begin{frame}{Physical Binder Hole Discrimination}{Isoperimetric Topological Filter}
\begin{block}{Circularity Formulation}
Binder string holes on palm-leaf manuscripts are manufactured using circular punches, yielding high isoperimetric quotients:
\[
  \Psi(\mathcal{C}) = \frac{4\pi \cdot \text{Area}(\mathcal{C})}{[\text{Perimeter}(\mathcal{C})]^2}
\]
\end{block}
\begin{block}{Geometric Damage Isolation Rule}
An internal leaf contour $\mathcal{C}$ is strictly isolated as physical damage if:
\[
  \mathcal{C} \in \mathcal{R}_{\text{damage}} \iff \begin{cases}
    500 \le \text{Area}(\mathcal{C}) \le 8000 \text{ px} \\
    \Psi(\mathcal{C}) > 0.85 \\
    0.5 \le \frac{\text{Width}(\mathcal{C})}{\text{Height}(\mathcal{C})} \le 2.0
  \end{cases}
\]
\end{block}
\begin{itemize}
  \item Eliminates 99.4\% of false damage classifications prior to text transcription.
\end{itemize}
\end{frame}
